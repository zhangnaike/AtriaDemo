    var yRange = yMax - yMin;

    function X(idx) { return C.padL + (plotW * idx) / (n - 1); }
    function Y(v)   { return C.padT + plotH * (1 - (v - yMin) / yRange); }

    /* defs：面积 + 折线渐变 */
    var defs = svgEl("defs", {});
    var gArea = svgEl("linearGradient", { id: "gArea", x1: "0", y1: "0", x2: "0", y2: "1" });
    gArea.appendChild(svgEl("stop", { offset: "0%", "stop-color": "#F97316", "stop-opacity": 0.38 }));
    gArea.appendChild(svgEl("stop", { offset: "55%", "stop-color": "#FBBF24", "stop-opacity": 0.14 }));
    gArea.appendChild(svgEl("stop", { offset: "100%", "stop-color": "#FBBF24", "stop-opacity": 0 }));
    defs.appendChild(gArea);
    var gLine = svgEl("linearGradient", { id: "gLine", x1: "0", y1: "0", x2: "1", y2: "0" });
    gLine.appendChild(svgEl("stop", { offset: "0%", "stop-color": "#FBBF24" }));
    gLine.appendChild(svgEl("stop", { offset: "100%", "stop-color": "#F97316" }));
    defs.appendChild(gLine);
    svg.appendChild(defs);

    /* 水平网格 + Y 轴刻度 */
    var ticks = 4;
    for (i = 0; i <= ticks; i++) {
      var yv = C.padT + (plotH * i) / ticks;
      var v = yMax - (yRange * i) / ticks;
      svg.appendChild(svgEl("line", {
        x1: C.padL, x2: C.W - C.padR, y1: yv, y2: yv,
        stroke: "rgba(255,255,255,0.06)", "stroke-width": 1
      }));
      var yt = svgEl("text", {
        x: C.padL - 10, y: yv + 4, "text-anchor": "end",
        fill: "#8A8A96", "font-size": 10.5
      });
      yt.textContent = v >= 1000 ? (v / 1000).toFixed(1) + "k" : Math.round(v);
      svg.appendChild(yt);
    }

    /* X 轴刻度（按密度抽稀，最多 8 个） */
    var step = Math.ceil(n / 8);
    for (i = 0; i < n; i += step) {
      var xt = svgEl("text", {
        x: X(i), y: C.H - C.padB + 20, "text-anchor": "middle",
        fill: "#8A8A96", "font-size": 10.5
      });
      xt.textContent = series[i].label;
      svg.appendChild(xt);
    }

    /* 折线 / 面积路径 */
    var linePath = "", i2, x, y;
    for (i2 = 0; i2 < n; i2++) {
      x = X(i2); y = Y(series[i2].active);
      linePath += (i2 === 0 ? "M" : "L") + x.toFixed(1) + " " + y.toFixed(1);
    }
    var areaPath = linePath +
      "L" + X(n - 1).toFixed(1) + " " + (C.padT + plotH) +
      "L" + X(0).toFixed(1) + " " + (C.padT + plotH) + "Z";

    svg.appendChild(svgEl("path", { d: areaPath, fill: "url(#gArea)" }));

    /* 基准虚线：期间均值 */
    var avg = values.reduce(function (a, b) { return a + b; }, 0) / n;
    svg.appendChild(svgEl("line", {
      x1: C.padL, x2: C.W - C.padR, y1: Y(avg), y2: Y(avg),
      stroke: "rgba(255,255,255,0.22)", "stroke-width": 1, "stroke-dasharray": "5 5"
    }));

    /* 主折线：底层光晕 + 主线 */
    svg.appendChild(svgEl("path", {
      d: linePath, fill: "none", stroke: "url(#gLine)",
      "stroke-width": 6, "stroke-linecap": "round", "stroke-linejoin": "round", opacity: 0.18
    }));
    svg.appendChild(svgEl("path", {
      d: linePath, fill: "none", stroke: "url(#gLine)",
      "stroke-width": 2.6, "stroke-linecap": "round", "stroke-linejoin": "round"
    }));

    /* hover 交互层：竖线 + 光晕 + 锚点 */
    var hover = svgEl("g", { opacity: 0 });
    var hLine = svgEl("line", {
      y1: C.padT, y2: C.padT + plotH,
      stroke: "rgba(255,255,255,0.28)", "stroke-width": 1, "stroke-dasharray": "3 3"
    });
    var halo = svgEl("circle", { r: 9, fill: "#F97316", opacity: 0.22 });
    var dot = svgEl("circle", { r: 4.6, fill: "#0A0A0F", stroke: "#FBBF24", "stroke-width": 2.4 });
    hover.appendChild(hLine);
    hover.appendChild(halo);
    hover.appendChild(dot);
    svg.appendChild(hover);

    var overlay = svgEl("rect", {
      x: C.padL, y: C.padT, width: plotW, height: plotH, fill: "transparent"
    });
    svg.appendChild(overlay);

    var tip = $("lineTooltip");

    function showAt(idx) {
      var px = X(idx), py = Y(series[idx].active);
      hover.setAttribute("opacity", 1);
      hLine.setAttribute("x1", px);
      hLine.setAttribute("x2", px);
      halo.setAttribute("cx", px); halo.setAttribute("cy", py);
      dot.setAttribute("cx", px);  dot.setAttribute("cy", py);

      var wrap = $("lineWrap").getBoundingClientRect();
      var sx = wrap.width / C.W, sy = wrap.height / C.H;
      tip.innerHTML = '<span class="t-label">' + series[idx].label +
        " · 活跃用户</span><br><b>" + fmt(series[idx].active) + "</b> 人";
      tip.style.left = (px * sx) + "px";
      tip.style.top = (py * sy) + "px";
      tip.classList.add("show");
    }

    function hide() {
      hover.setAttribute("opacity", 0);
      tip.classList.remove("show");
    }

    function nearestIndex(clientX) {
      var rect = svg.getBoundingClientRect();
      var svgX = (clientX - rect.left) * (C.W / rect.width);
      var idx = Math.round(((svgX - C.padL) / plotW) * (n - 1));
      return Math.max(0, Math.min(n - 1, idx));
    }

    overlay.addEventListener("mousemove", function (ev) { showAt(nearestIndex(ev.clientX)); });
    overlay.addEventListener("mouseleave", hide);
    overlay.addEventListener("touchmove", function (ev) {
      ev.preventDefault();
      showAt(nearestIndex(ev.touches[0].clientX));
    }, { passive: false });
    overlay.addEventListener("touchend", hide);
  }

  /* ================= 环形图 ================= */
  function renderDonut(sources) {
    var svg = $("donutChart");
    svg.innerHTML = "";
    var total = 0, i;
    for (i = 0; i < sources.length; i++) total += sources[i].value;

    var R = 44, CENTER = 60, circ = 2 * Math.PI * R, offset = 0;

    svg.appendChild(svgEl("circle", {
      cx: CENTER, cy: CENTER, r: R, fill: "none",
      stroke: "rgba(255,255,255,0.06)", "stroke-width": 13
    }));

    var segs = [];
    for (i = 0; i < sources.length; i++) {
      var s = sources[i];
      var len = (s.value / total) * circ;
      var arc = svgEl("circle", {
        cx: CENTER, cy: CENTER, r: R, fill: "none",
        stroke: s.color, "stroke-width": 13, "stroke-linecap": "round",
        "stroke-dasharray": (len - 2.5) + " " + (circ - len + 2.5),
        "stroke-dashoffset": -offset,
        opacity: 0.92
      });
      arc.setAttribute("class", "donut-segment");
      svg.appendChild(arc);
      segs.push(arc);
      offset += len;
    }

    function setCenter(idx) {
      $("donutValue").textContent = Math.round(sources[idx].value / total * 100) + "%";
      $("donutLabel").textContent = sources[idx].name;
      $("donutValue").style.color = sources[idx].color;
      for (var k = 0; k < segs.length; k++) {
        segs[k].style.opacity = (k === idx ? 1 : 0.3);
        segs[k].style.strokeWidth = (k === idx ? 16 : 13);
      }
      var items = $("donutLegend").children;
      for (var m = 0; m < items.length; m++) items[m].classList.toggle("dim", m !== idx);
    }

    /* 图例 */
    var legend = $("donutLegend");
    legend.innerHTML = "";
    var frag = document.createDocumentFragment();
    for (i = 0; i < sources.length; i++) {
      (function (idx) {
        var s = sources[idx];
        var row = document.createElement("div");
        row.className = "item";
        row.innerHTML =
          '<i style="background:' + s.color + '"></i>' +
          '<span class="name">' + s.name + "</span>" +
          '<span class="pct">' + Math.round(s.value / total * 100) + "%</span>";

        function on()  { setCenter(idx); }
        function off() { setCenter(0); }
        row.addEventListener("mouseenter", on);
        row.addEventListener("mouseleave", off);
        segs[idx].addEventListener("mouseenter", on);
        segs[idx].addEventListener("mouseleave", off);
        frag.appendChild(row);
      })(i);
    }
    legend.appendChild(frag);
    setCenter(0);
  }

  /* ================= 订单表格 ================= */
  function renderOrders(seed) {
    var tbody = $("orderBody");
    tbody.innerHTML = "";
    var frag = document.createDocumentFragment();
    for (var i = 0; i < 8; i++) {
      var cu = CUSTOMERS[(seed + i * 3) % CUSTOMERS.length];
      var st = STATUS[(seed + i) % STATUS.length];
      var price = PLAN_PRICE[cu[1]];
      var qty = 1 + ((seed + i) % 3);

      var tr = document.createElement("tr");
      tr.innerHTML =
        '<td class="order-id">#LM-' + (20251000 + (seed + i) * 137) + "</td>" +
        '<td><div class="customer">' +
          '<span class="avatar" style="background:linear-gradient(135deg,' +
          AVATAR_G[(seed + i) % 4] + ')">' + cu[0].slice(0, 1) + "</span>" +
          cu[0] +
        "</div></td>" +
        "<td>" + cu[1] + "</td>" +
        '<td style="text-align:right" class="amount">¥' + fmt(price * qty) + "</td>" +
        '<td><span class="badge ' + st[0] + '"><i></i>' + st[1] + "</span></td>";
      frag.appendChild(tr);
    }
    tbody.appendChild(frag);
  }

  /* ================= 渲染当前时间范围 ================= */
  function renderRange(key, animate) {
    currentRange = key;
    var data = RANGES[key];
    var k = data.kpis;

    $("rangeLabel").textContent = data.label;

    if (animate) {
      animateValue($("kpiActive"),  k.active[1], 1400, fmt);
      animateValue($("kpiRevenue"), k.revenue[1], 1400, fmt);
      animateValue($("kpiConvert"), k.convert[1], 1200, function (v) { return v.toFixed(2); });
    } else {
      $("kpiActive").textContent = fmt(k.active[1]);
      $("kpiRevenue").textContent = fmt(k.revenue[1]);
      $("kpiConvert").textContent = k.convert[1].toFixed(2);
    }
    renderDelta($("kpiActiveDelta"), k.active[2]);
    renderDelta($("kpiRevenueDelta"), k.revenue[2]);
    renderDelta($("kpiConvertDelta"), k.convert[2]);

    renderLineChart(data.series);
    renderDonut(data.sources);

    var buttons = $("rangeSwitch").querySelectorAll("button");
    for (var i = 0; i < buttons.length; i++) {
      buttons[i].classList.toggle("active", buttons[i].getAttribute("data-range") === key);
    }
  }

  /* ================= 事件绑定 ================= */
  $("rangeSwitch").addEventListener("click", function (ev) {
    var btn = ev.target.closest("button");
    if (!btn) return;
    var key = btn.getAttribute("data-range");
    if (key !== currentRange) renderRange(key, true);
  });

  /* ================= 伪实时推送（demo 效果） ================= */
  var tick = 0;
  setInterval(function () {
    tick++;
    /* 今日活跃小幅抖动 + 青色闪烁高亮 */
    var base = RANGES[currentRange].kpis.active[1];
    var jitter = ((tick % 3) * 2 + 1) * 37;
    var el = $("kpiActive");
    el.textContent = fmt(base + jitter);
    el.classList.remove("flash");
    void el.offsetWidth; /* 重置动画，使每轮都能触发 */
    el.classList.add("flash");

    /* 订单表每两轮刷新一次 */
    if (tick % 2 === 0) renderOrders(tick * 5);
  }, 4000);

  /* ================= 初始化 ================= */
  renderOrders(0);
  renderRange("14d", true);
})();
