<script>
(function () {
  "use strict";

  /* 1. nav turns solid on scroll */
  var nav = document.getElementById("nav");
  var ticking = false;
  function onScroll() {
    if (ticking) return;
    ticking = true;
    window.requestAnimationFrame(function () {
      nav.classList.toggle("solid", window.scrollY > 24);
      ticking = false;
    });
  }
  window.addEventListener("scroll", onScroll, { passive: true });
  onScroll();

  /* 2. typewriter */
  var TEXT = "The harbor was already awake at five. Fishing boats knocked against the quay like a city finding its rhythm, and the first light came in low and gold, painting the plaster walls of the old town the color of warm bread. I drank my coffee standing up, the way the locals do, and decided that a place this generous with its mornings deserved at least one more day of mine.";

  var typeEl = document.getElementById("type");
  var counterEl = document.getElementById("counter");
  var caretEl = document.getElementById("caret");
  var i = 0;

  function countWords(s) {
    var t = s.replace(/\s+/g, " ").trim();
    return t.length ? t.split(" ").length : 0;
  }

  function typeStep() {
    if (i >= TEXT.length) {
      counterEl.textContent = countWords(TEXT) + " words, done";
      caretEl.style.animationDuration = "0.5s";
      return;
    }
    i++;
    typeEl.textContent = TEXT.slice(0, i);
    counterEl.textContent = countWords(TEXT.slice(0, i)) + " words";
    var ch = TEXT.charAt(i - 1);
    var delay = /[,.;:]/.test(ch) ? 240 : (ch === " " ? 14 : 26 + Math.random() * 46);
    window.setTimeout(typeStep, delay);
  }

  window.setTimeout(typeStep, 1100);

  /* 3. scroll reveal via IntersectionObserver */
  var reveals = document.querySelectorAll(".reveal");
  if ("IntersectionObserver" in window) {
    var io = new IntersectionObserver(function (entries) {
      entries.forEach(function (entry) {
        if (entry.isIntersecting) {
          entry.target.classList.add("in");
          io.unobserve(entry.target);
        }
      });
    }, { threshold: 0.14, rootMargin: "0px 0px -8% 0px" });
    reveals.forEach(function (el) { io.observe(el); });
  } else {
    reveals.forEach(function (el) { el.classList.add("in"); });
  }
})();
</script>
